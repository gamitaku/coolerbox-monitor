import machine import time import network import urequests # ========================================== # 1. 接続 & 設定情報 # ========================================== WIFI_SSID = "F660P-zbQH-G" # 2.4GHz帯SSID WIFI_PASS = "kesy4eb7ysyb4" AMBIENT_CHANNEL_ID = "102675" AMBIENT_WRITE_KEY = "408d25019deace59" SEND_INTERVAL_SEC = 300 # 5分間隔 (300秒) # ========================================== # 2. ハードウェア初期化 (センサー & WDT) # ========================================== # 内蔵温度センサー (ADC4) adc_temp = machine.ADC(4) # VSYS 電圧測定 (ADC29 / PIN29) adc_vsys = machine.ADC(29) # ウォッチドッグタイマー (上限8秒に設定) wdt = machine.WDT(timeout=8000) # ========================================== # 3. 計測用関数 # ========================================== def read_temperature(): """内蔵温度センサーから温度(℃)を取得""" adc_value = adc_temp.read_u16() voltage = adc_value * (3.3 / 65535) temperature = 27 - (voltage - 0.706) / 0.001721 return round(temperature, 1) def read_vsys_voltage(): """VSYSピンからバッテリー/電源電圧(V)を取得""" adc_value = adc_vsys.read_u16() vsys_voltage = adc_value * (3.3 * 3 / 65535) return round(vsys_voltage, 2) def get_wifi_rssi(): """Wi-Fiの電波強度(RSSI dBm)を取得""" wlan = network.WLAN(network.STA_IF) if wlan.isconnected(): try: return wlan.status('rssi') except: return -99 return -99 # ========================================== # 4. Wi-Fi 接続処理 # ========================================== def connect_wifi(): wlan = network.WLAN(network.STA_IF) wlan.active(False) # 一度リセット time.sleep(0.5) wlan.active(True) if not wlan.isconnected(): print(f"Wi-Fi接続中: {WIFI_SSID}...") wlan.connect(WIFI_SSID, WIFI_PASS) max_wait = 20 while max_wait > 0: wdt.feed() if wlan.status() >= 3: break max_wait -= 1 time.sleep(1) if wlan.isconnected(): print(f"Wi-Fi接続成功! IP: {wlan.ifconfig()[0]}, RSSI: {get_wifi_rssi()} dBm") return True else: print("Wi-Fi接続失敗") return False # ========================================== # 5. Ambient データ送信処理 # ========================================== def send_to_ambient(temp, volts, rssi): url = f"https://ambidata.io/api/v2/channels/{AMBIENT_CHANNEL_ID}/data" headers = {'Content-Type': 'application/json'} payload = { "writeKey": AMBIENT_WRITE_KEY, "d1": temp, "d2": volts, "d3": rssi } try: response = urequests.post(url, json=payload, headers=headers) print(f"[{time.ticks_ms()//1000}s] 送信完了: 温度={temp}℃, 電圧={volts}V, RSSI={rssi}dBm") response.close() return True except Exception as e: print(f"Ambient送信エラー: {e}") return False # ========================================== # 6. メインループ # ========================================== def main(): print("=== 保冷BOX・電源監視システム 起動 ===") connect_wifi() last_send_time = 0 while True: wdt.feed() current_time = time.time() if current_time - last_send_time >= SEND_INTERVAL_SEC or last_send_time == 0: wlan = network.WLAN(network.STA_IF) if not wlan.isconnected(): print("Wi-Fi切断を検知。再接続試行中...") connect_wifi() if wlan.isconnected(): temp = read_temperature() volts = read_vsys_voltage() rssi = get_wifi_rssi() if send_to_ambient(temp, volts, rssi): last_send_time = current_time time.sleep(1) if __name__ == "__main__": main()
